#### Tabela `despesas`
```sql
CREATE TABLE despesas (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    descricao TEXT NOT NULL,
    categoria TEXT NOT NULL, -- Ex: 'Material Didático', 'Alimentação', 'Liturgia', 'Outros'
    valor NUMERIC(10,2) NOT NULL,
    data_despesa DATE NOT NULL DEFAULT CURRENT_DATE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);
```

#### Tabela `entradas_extras`
```sql
CREATE TABLE entradas_extras (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    descricao TEXT NOT NULL,
    origem TEXT NOT NULL, -- Ex: 'Taxa Extra', 'Doação', 'Oferta', 'Inscrição Retiro'
    valor NUMERIC(10,2) NOT NULL,
    data_entrada DATE NOT NULL DEFAULT CURRENT_DATE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);
```

#### Tabela `encontros`
```sql
CREATE TABLE encontros (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    data_encontro DATE NOT NULL UNIQUE,
    tema TEXT,
    observacao TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);
```

#### Tabela `presencas`
```sql
CREATE TABLE presencas (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    encontro_id BIGINT REFERENCES encontros(id) ON DELETE CASCADE,
    crismando_id BIGINT REFERENCES crismandos(id) ON DELETE CASCADE,
    status TEXT NOT NULL CHECK (status IN ('PRESENTE', 'FALTA', 'JUSTIFICADO')),
    observacao TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    UNIQUE(encontro_id, crismando_id)
);
```


### 2.2 Função de Disparo de Texto (`js/evolution.js` ou módulo equivalente)
```javascript
async function enviarTextoEvolutionAPI(telefone, mensagem) {
    const numLimpo = telefone.replace(/\D/g, '');
    const numFormatado = numLimpo.startsWith('55') ? numLimpo : '55' + numLimpo;

    const url = `${ENV.EVOLUTION_API_URL}/message/sendText/${ENV.EVOLUTION_INSTANCE_NAME}`;
    
    const payload = {
        number: numFormatado,
        options: {
            delay: 1200,
            presence: "composing",
            linkPreview: false
        },
        textMessage: {
            text: mensagem
        }
    };

    const response = await fetch(url, {
        method: 'POST',
        headers: {
            'Content-Type': 'application/json',
            'apikey': ENV.EVOLUTION_API_KEY
        },
        body: JSON.stringify(payload)
    });

    return response.ok;
}
```
